97def parse_line(line):
98
107 m = LINE_PATTERN.search(line)
108 if not m:
109 return None
110 freq = float(m.group(1))
111 samples = [int(x.strip()) for x in m.group(2).split(",") if x.strip()]
112 avg = int(m.group(3))
113 return freq, samples, avg

116def parse_scan_info(line):
117
127 m = INFO_TOTAL.search(line)
128 if m:
129 return {"tipo": "Total", "xmin": 80.0, "xmax": 108.0}
130 m = INFO_CENTRAL.search(line)
131 if m:
132 centro = int(m.group(1)) / 100.0
133 passo = int(m.group(2)) / 100.0
134 xmin = max(80.0, centro - 10 * passo)
135 xmax = min(108.0, centro + 10 * passo)
136 return {"tipo": "Central", "xmin": round(xmin, 2),
137 "xmax": round(xmax, 2), "centro": centro}
138 return None

141def find_arduino_port():
142
151 ports = serial.tools.list_ports.comports()
152 keywords = ["arduino", "ch340", "cp210", "ftdi", "usb serial", "esp32", "esp"]
153 for p in ports:
154 desc = (p.description or "").lower()
155 mfr = (p.manufacturer or "").lower()
156 if any(k in desc or k in mfr for k in keywords):
157 return p.device
158 if ports:
159 return ports[0].device
160 return None

164

597 def update(_frame):
598
608
609 # 1. Process commands from dashboard / keyboard
610 while state["cmd_queue"]:
611 cmd = state["cmd_queue"].pop(0)
612 if cmd == "PAUSE":
613 state["paused"] = not state["paused"]
614 status_text.set_text("PAUSED - zoom/pan active" if state["paused"] else "LIVE")
615 status_text.set_color("#cc7700" if state["paused"] else "#2a9d2a")
616 elif cmd == "CHANGE_COLOUR":
617 state["colour_idx"] = (state["colour_idx"] + 1) % len(COLOURS)
618 update_colour_label()
619 elif cmd == "RESET":
620 do_reset()
621 elif cmd == "SAVE":
622 do_save(fig, state["scans"], state["raw_lines"])
623 elif cmd == "FREQ_UP":
624 send(f"FREQ:{min(state['frequency']+10, 10800)}")
625 elif cmd == "FREQ_DOWN":
626 send(f"FREQ:{max(state['frequency']-10, 8000)}")
627 else:
628 send(cmd) # VOL+, VOL-, MUTE, FREQ:xxxx, SCAN_*
629
630 # 2. Read serial
631 while ser.in_waiting:
632 raw = ser.readline()
633 try:
634 text = raw.decode("utf-8", errors="replace").strip()
635 except Exception:
636 continue
637 if not text:
638 continue
639 print(f" {text}")
640 state["raw_lines"].append(text)
641
642 info = parse_scan_info(text)
643 if info:
644 state["scan_info"] = info
645 apply_scan_info(info)
646 continue
647
648 m = VOL_PATTERN.match(text)
649 if m:
650 state["volume"] = int(m.group(1))
651 continue
652
653 m = MUTE_PATTERN.match(text)
654 if m:
655 state["muted"] = (m.group(1) == "ON")
656 continue
657
658 m = FREQ_PATTERN.match(text)
659 if m:
660 state["frequency"] = int(m.group(1))
661 continue
662
663 result = parse_line(text)
664 if result:
665 freq, samples, avg = result
666 scan = get_or_create_scan()
667 scan["freqs"].append(freq)
668 scan["avgs"].append(avg)
669 scan["all_samples"].append(samples)
670
671 # 3. Refresh dashboard
672 try:
673 dash.refresh(state)
674 dash.update_idletasks()
675 except Exception:
676 pass
677
678 # 4. Redraw graph
679 if state["paused"]:
680 return
681
682 for scan in state["scans"]:
683 if len(scan["freqs"]) < 2:
684 continue
685 f = np.array(scan["freqs"])
686 a = np.array(scan["avgs"])
687 mn = np.array([min(s) for s in scan["all_samples"]])
688 mx = np.array([max(s) for s in scan["all_samples"]])
689 if scan["fill_obj"] is not None:
690 scan["fill_obj"].remove()
691 scan["fill_obj"] = ax.fill_between(
692 f, mn, mx, color=scan["fill_col"], alpha=0.20)
693 scan["line_obj"].set_data(f, a)
694 if mx.max() + 5 > ax.get_ylim()[1]:
695 ax.set_ylim(0, mx.max() + 5)
696
697 ani = FuncAnimation(fig, update, interval=200, cache_frame_data=False)
698 plt.show()
699
700 NavigationToolbar2.save_figure = original_save
701 ser.close()
702 try:
703 dash.destroy()
704 except Exception:
705 pass
706 print("[OK] Session ended.")
